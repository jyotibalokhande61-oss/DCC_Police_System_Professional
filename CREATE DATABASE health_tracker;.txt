CREATE DATABASE health_tracker;

USE health_tracker;

CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100),
    password VARCHAR(100)
);

CREATE TABLE health_data (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    date DATE,
    weight FLOAT,
    steps INT,
    water INT,
    sleep INT
);
